# 01 Create Tables

```sql
-- PROPOSED PostgreSQL schema for CMS OpenAPI v0.6.3.
-- For a NEW empty database. Do not run as a migration over existing tables.
-- Run 01_create_tables.sql, then 02_indexes.sql, then 03_seed_unknown.sql.

-- 1/5: authors
CREATE TABLE authors (
    id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    CONSTRAINT authors_name_not_blank CHECK (char_length(btrim(name)) > 0)
);

-- 2/5: categories
CREATE TABLE categories (
    id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    is_system BOOLEAN NOT NULL DEFAULT FALSE,
    CONSTRAINT categories_name_not_blank CHECK (char_length(btrim(name)) > 0)
);

-- The API registers externally hosted media references, not binary file contents.
-- URLs are limited to 2048 characters, matching the OpenAPI maxLength.

-- 3/5: media
CREATE TABLE media (
    id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    file_name VARCHAR(255) NOT NULL,
    media_type VARCHAR(20) NOT NULL,
    url VARCHAR(2048) NOT NULL,
    CONSTRAINT media_file_name_not_blank CHECK (char_length(btrim(file_name)) > 0),
    CONSTRAINT media_type_allowed CHECK (media_type IN ('image', 'video', 'audio')),
    CONSTRAINT media_url_not_blank CHECK (char_length(btrim(url)) > 0)
);

-- A preview is stored in the same row as its article.
-- published_at = first publication time WITHIN THE CURRENT ACTIVATION CYCLE.

-- 4/5: articles;
CREATE TABLE articles (
    id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    slug VARCHAR(255) NOT NULL,
    author_id INTEGER NOT NULL,
    category_id INTEGER NOT NULL,
    main_image_id INTEGER NOT NULL,
    content JSONB NOT NULL,
    preview_image_id INTEGER NOT NULL,
    preview_content JSONB NOT NULL,
    status VARCHAR(20) DEFAULT 'draft',
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    published_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    CONSTRAINT articles_title_not_blank CHECK (char_length(btrim(title)) > 0),
    CONSTRAINT articles_slug_not_blank CHECK (char_length(btrim(slug)) > 0),
    CONSTRAINT articles_slug_unique UNIQUE (slug),
    CONSTRAINT articles_publication_state_check CHECK (
        (is_active = FALSE AND status IS NULL AND published_at IS NULL)
        OR (is_active = TRUE AND status IS NOT NULL AND status = 'draft' AND published_at IS NULL)
        OR (is_active = TRUE AND status IS NOT NULL AND status = 'published' AND published_at IS NOT NULL)
    ),
    CONSTRAINT articles_content_nonempty_array CHECK (
        CASE WHEN jsonb_typeof(content) = 'array'
             THEN jsonb_array_length(content) > 0 ELSE FALSE END
    ),
    CONSTRAINT articles_preview_nonempty_array CHECK (
        CASE WHEN jsonb_typeof(preview_content) = 'array'
             THEN jsonb_array_length(preview_content) > 0 ELSE FALSE END
    ),
    CONSTRAINT articles_author_fk FOREIGN KEY (author_id) REFERENCES authors(id) ON DELETE RESTRICT,
    CONSTRAINT articles_category_fk FOREIGN KEY (category_id) REFERENCES categories(id) ON DELETE RESTRICT,
    CONSTRAINT articles_main_image_fk FOREIGN KEY (main_image_id) REFERENCES media(id) ON DELETE RESTRICT,
    CONSTRAINT articles_preview_image_fk FOREIGN KEY (preview_image_id) REFERENCES media(id) ON DELETE RESTRICT
);

-- 5/5: hero banners
CREATE TABLE hero_banners (
    id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    image_id INTEGER NOT NULL,
    content JSONB NOT NULL,
    status VARCHAR(20) DEFAULT 'draft',
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    published_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    CONSTRAINT hero_banners_publication_state_check CHECK (
        (is_active = FALSE AND status IS NULL AND published_at IS NULL)
        OR (is_active = TRUE AND status IS NOT NULL AND status = 'draft' AND published_at IS NULL)
        OR (is_active = TRUE AND status IS NOT NULL AND status = 'published' AND published_at IS NOT NULL)
    ),
    CONSTRAINT hero_banners_content_nonempty_array CHECK (
        CASE WHEN jsonb_typeof(content) = 'array'
             THEN jsonb_array_length(content) > 0 ELSE FALSE END
    ),
    CONSTRAINT hero_banners_image_fk FOREIGN KEY (image_id) REFERENCES media(id) ON DELETE RESTRICT
);
```
